@login_required
@require_GET
def api_events(request):
    start = request.GET.get("start")
    end = request.GET.get("end")
    company = get_user_company(request)
    qs = Appointment.objects.filter(company=company).select_related("client", "resource", "service") if company else Appointment.objects.none()
    if start:
        qs = qs.filter(end_time__gte=start)
    if end:
        qs = qs.filter(start_time__lte=end)
    events = []
    for app in qs:
        client_name = app.client.name if app.client else "Interno"
        events.append({
            "id": app.id,
            "title": f"{app.title} ({client_name})",
            "start": app.start_time.isoformat() if app.start_time else None,
            "end": app.end_time.isoformat() if app.end_time else None,
            "allDay": app.is_all_day,
            "backgroundColor": app.color or "#3b82f6",
            "borderColor": app.color or "#3b82f6",
            "extendedProps": {
                "type": app.get_appointment_type_display(),
                "status": app.get_status_display(),
                "status_code": app.status,
                "resource_id": app.resource_id,
                "client": client_name,
                "description": app.description,
            },
        })
    return JsonResponse(events, safe=False)


@login_required
def appointment_list(request):
    q = request.GET.get("q", "")
    status = request.GET.get("status", "")
    date_from = request.GET.get("date_from", "")
    date_to = request.GET.get("date_to", "")

    company = get_user_company(request)
    qs = Appointment.objects.filter(company=company).select_related("client", "resource", "employee").order_by("-start_time") if company else Appointment.objects.none()
    if q:
        qs = qs.filter(Q(title__icontains=q) | Q(client__name__icontains=q) | Q(description__icontains=q))
    if status:
        qs = qs.filter(status=status)
    if date_from:
        qs = qs.filter(start_time__date__gte=date_from)
    if date_to:
        qs = qs.filter(start_time__date__lte=date_to)

    paginator = Paginator(qs, 30)
    page = paginator.get_page(request.GET.get("page"))
    return render(request, "appointments/appointment_list.html", {
        "page_obj": page,
        "q": q, "status": status, "date_from": date_from, "date_to": date_to,
        "total": qs.count(),
        "status_choices": Appointment.STATUS_CHOICES,
    })

@login_required
def appointment_status(request, pk):
    appt = get_object_or_404(Appointment, pk=pk)
    new_status = request.POST.get("status")
    if new_status in dict(Appointment.STATUS_CHOICES):
        old_st = appt.status
        appt.status = new_status
        appt.save()
        messages.success(request, f"Status promijenjen u: {appt.get_status_display()}")
        if new_status != old_st and new_status in ("confirmed", "cancelled", "completed", "no_show"):
            try:
                from appointments.tasks import notify_appointment_status_change
                notify_appointment_status_change.delay(appt.id, new_status)
            except Exception:
                pass
    return redirect("appointments:list")


@csrf_exempt
@login_required
def api_appointments(request, pk=None):
    if request.method in ["POST", "PUT"]:
        try:
            data = json.loads(request.body) if request.content_type == "application/json" else request.POST
        except Exception:
            data = request.POST
        if pk:
            appointment = get_object_or_404(Appointment, pk=pk)
            form = AppointmentForm(data, instance=appointment)
        else:
            form = AppointmentForm(data)
        if form.is_valid():
            appointment = form.save(commit=False)
            if not getattr(appointment, "employee_id", None):
                appointment.employee = request.user
            appointment.save()
            return JsonResponse({"status": "success", "id": appointment.id})
        return JsonResponse({"status": "error", "errors": form.errors}, status=400)
    elif request.method == "DELETE" and pk:
        appointment = get_object_or_404(Appointment, pk=pk)
        appointment.delete()
        return JsonResponse({"status": "success"})
    return JsonResponse({"status": "invalid method"}, status=405)

@require_GET
def api_available_slots(request):
    date_str = request.GET.get("date")
    service_id = request.GET.get("service_id")
    if not date_str or not service_id:
        return JsonResponse({"error": "Nedostaju parametri"}, status=400)
    target_date = parse_date(date_str)
    if not target_date or target_date < datetime.now().date():
        return JsonResponse({"available": [], "unavailable": [], "slots": []})
    service = get_object_or_404(BookingService, id=service_id)
    weekday = target_date.weekday()
    resources = BookingResource.objects.filter(resource_type=service.resource_type, is_active=True)
    available_slots = set()
    unavailable_slots = set()
    service_duration = service.duration.total_seconds() / 60
    for resource in resources:
        wh = WorkingHours.objects.filter(resource=resource, day_of_week=weekday, is_day_off=False).first()
        if not wh:
            continue
        start_dt = datetime.combine(target_date, wh.start_time)
        end_dt = datetime.combine(target_date, wh.end_time)
        existing_apps = list(Appointment.objects.filter(
            resource=resource, start_time__date=target_date, status__in=["scheduled", "confirmed"]
        ))
        current_slot = start_dt
        while current_slot + timedelta(minutes=service_duration) <= end_dt:
            slot_end = current_slot + timedelta(minutes=service_duration)
            overlap = any(
                current_slot < a.end_time.replace(tzinfo=None) and slot_end > a.start_time.replace(tzinfo=None)
                for a in existing_apps
            )
            if target_date == datetime.now().date() and current_slot.time() < datetime.now().time():
                overlap = True
            slot_str = current_slot.strftime("%H:%M")
            if not overlap:
                available_slots.add(slot_str)
            else:
                unavailable_slots.add(slot_str)
            current_slot += timedelta(minutes=service.slot_duration_minutes)
    avail = sorted(list(available_slots))
    unavail = sorted(list(unavailable_slots - available_slots))
    return JsonResponse({"available": avail, "unavailable": unavail, "slots": avail})


@csrf_exempt
@transaction.atomic
def api_submit_public_booking(request):
    if request.method != "POST":
        return JsonResponse({"error": "Invalid request"}, status=400)
    # Zahtjeva autentikaciju
    if not request.user.is_authenticated:
        return JsonResponse({"error": "auth_required", "message": "Morate biti prijavljeni za rezervaciju."}, status=401)
    try:
        data = json.loads(request.body)
        service = get_object_or_404(BookingService, id=data["service_id"])
        target_date = parse_date(data["date"])
        start_time = datetime.strptime(f"{data['date']} {data['time']}", "%Y-%m-%d %H:%M")
        end_time = start_time + timedelta(minutes=service.duration.total_seconds() / 60)
        # Klijent — pokušaj vezati za korisnika ako postoji partner
        email = data.get("email") or request.user.email
        name  = data.get("name")  or request.user.get_full_name() or request.user.username
        # Get company from service resource or first active company
        import uuid as _uuid

        from companies.models import Company
        booking_company = Company.objects.filter(is_platform_owner=True).first() or Company.objects.first()
        # Try existing partner by email first, then create
        client = Partner.objects.filter(email=email, company=booking_company).first()
        if not client:
            client = Partner.objects.filter(email=email).first()
        if not client:
            # Generate unique placeholder OIB to avoid unique_together constraint
            placeholder_oib = f"B{_uuid.uuid4().hex[:10].upper()}"
            client = Partner(
                name=name, email=email, company=booking_company,
                phone=data.get("phone", ""), partner_type="KUPAC",
                oib=placeholder_oib,
            )
            client.save()
        resources = BookingResource.objects.filter(resource_type=service.resource_type, is_active=True)
        assigned_resource = None
        for resource in resources:
            if not Appointment.objects.filter(
                resource=resource, start_time__lt=end_time, end_time__gt=start_time,
                status__in=["scheduled", "confirmed"]
            ).exists():
                assigned_resource = resource
                break
        if not assigned_resource:
            return JsonResponse({"error": "Termin je zauzet. Odaberite drugi."}, status=400)
        appointment = Appointment.objects.create(
            company=booking_company,
            title=f"Online: {service.name}", client=client, service=service,
            resource=assigned_resource, start_time=start_time, end_time=end_time,
            appointment_type="service", status="scheduled", color="#10b981",
            total_price=service.price or 0,
            payment_method=data.get("payment_method", "invoice"),
            whatsapp_number=data.get("whatsapp", ""),
        )
        # ── Potvrda emailom ───────────────────────────────────────────────────
        try:
            from django.conf import settings

            from core.email_tasks import send_email_task
            dt_fmt = start_time.strftime("%d.%m.%Y u %H:%M")
            subject = f"Potvrda rezervacije — {service.name}"
            body = (
                f"Poštovani/a {name},\n\n"
                f"Vaša rezervacija je uspješno kreirana.\n\n"
                f"  Usluga:  {service.name}\n"
                f"  Termin:  {dt_fmt}\n"
                f"  Trajanje:{int(service.duration.total_seconds()//60)} min\n"
            )
            if service.price:
                body += f"  Cijena:  {service.price} EUR\n"
            body += (
                f"\nAko trebate otkazati ili promijeniti termin, kontaktirajte nas.\n\n"
                f"Hvala na povjerenju!\n"
            )
            send_email_task.delay(
                subject, body,
                getattr(settings, 'DEFAULT_FROM_EMAIL', 'noreply@example.com'),
                [email],
                # async via celery
            )
        except Exception:
            pass
        ics_url = f"/appointments/booking/ics/{appointment.id}/"
        # Notify staff about new booking
        try:
            from django.contrib.auth.models import User as _U

            from notifications.services import notify_bulk
            staff = _U.objects.filter(is_staff=True)
            notify_bulk(
                recipients=staff,
                notification_type='appointment_confirmed',
                title='Nova online rezervacija',
                message=f'{service.name} - {start_time.strftime("%d.%m.%Y %H:%M")} ({name})',
                url='/appointments/',
            )
        except Exception:
            pass
        # WhatsApp potvrda
        if appointment.whatsapp_number:
            try:
                from core.whatsapp_tasks import send_appointment_confirmation_whatsapp
                send_appointment_confirmation_whatsapp.delay(appointment.id)
            except Exception:
                pass
        return JsonResponse({"status": "success", "booking_id": appointment.id, "total_price": float(appointment.total_price or 0), "ics_url": ics_url})
    except Exception as e:
        return JsonResponse({"error": str(e)}, status=500)

@csrf_exempt
def api_booking_register(request):
    """AJAX registracija novog korisnika za booking portal"""
    if request.method != 'POST':
        return JsonResponse({"error": "Invalid method"}, status=405)
    try:
        from django.contrib.auth import get_user_model, login
        from django.contrib.auth.password_validation import validate_password
        from django.core.exceptions import ValidationError
        User = get_user_model()
        data = json.loads(request.body)
        username  = data.get('username', '').strip()
        email     = data.get('email', '').strip()
        password  = data.get('password', '')
        first     = data.get('first_name', '').strip()
        last      = data.get('last_name', '').strip()

        if not username or not password:
            return JsonResponse({"success": False, "error": "Korisničko ime i lozinka su obavezni."})
        if User.objects.filter(username=username).exists():
            return JsonResponse({"success": False, "error": "Korisničko ime je zauzeto."})
        if email and User.objects.filter(email=email).exists():
            return JsonResponse({"success": False, "error": "Email je već registriran."})
        # Password validation
        try:
            validate_password(password)
        except ValidationError as ve:
            return JsonResponse({"success": False, "error": " ".join(ve.messages)})

        user = User.objects.create_user(
            username=username, email=email, password=password,
            first_name=first, last_name=last
        )
        login(request, user)
        return JsonResponse({
            "success": True,
            "email": user.email,
            "full_name": user.get_full_name() or user.username
        })
    except Exception as e:
        return JsonResponse({"success": False, "error": str(e)})

# ── TJEDNI RASPORED SLOBODNIH TERMINA ──────────────────────────────────────────

@require_GET
def api_slots_week(request):
    service_id = request.GET.get("service_id")
    week_start_str = request.GET.get("week_start")
    if not service_id:
        return JsonResponse({"error": "service_id obavezan"}, status=400)
    service = get_object_or_404(BookingService, id=service_id, is_active=True)
    step = service.slot_duration_minutes
    dur  = service.duration_minutes
    from datetime import timedelta
    today = datetime.now().date()
    now_time = datetime.now().time()
    if week_start_str:
        try:
            week_start = parse_date(week_start_str)
        except Exception:
            week_start = today - timedelta(days=today.weekday())
    else:
        week_start = today - timedelta(days=today.weekday())
    DAYS_HR = ["Pon", "Uto", "Sri", "Čet", "Pet", "Sub", "Ned"]
    DAYS_FULL = ["Ponedjeljak", "Utorak", "Srijeda", "Četvrtak", "Petak", "Subota", "Nedjelja"]
    resources = BookingResource.objects.filter(resource_type=service.resource_type, is_active=True)
    days_data = []
    for day_offset in range(7):
        target_date = week_start + timedelta(days=day_offset)
        weekday = target_date.weekday()
        is_today = (target_date == today)
        is_past  = (target_date < today)
        all_available = set()
        all_unavailable = set()
        for resource in resources:
            wh = WorkingHours.objects.filter(
                resource=resource, day_of_week=weekday, is_day_off=False
            ).first()
            if not wh:
                continue
            start_dt = datetime.combine(target_date, wh.start_time)
            end_dt   = datetime.combine(target_date, wh.end_time)
            existing = list(Appointment.objects.filter(
                resource=resource, start_time__date=target_date,
                status__in=["scheduled", "confirmed"]
            ))
            current = start_dt
            while current + timedelta(minutes=dur) <= end_dt:
                slot_end = current + timedelta(minutes=dur)
                overlap = any(
                    current < a.end_time.replace(tzinfo=None) and slot_end > a.start_time.replace(tzinfo=None)
                    for a in existing
                )
                slot_str = current.strftime("%H:%M")
                if is_today and current.time() <= now_time:
                    overlap = True
                if not overlap:
                    all_available.add(slot_str)
                else:
                    all_unavailable.add(slot_str)
                current += timedelta(minutes=step)
        avail = sorted(all_available)
        unavail = sorted(all_unavailable - all_available)
        all_slots = sorted(set(avail) | set(unavail))
        slots = [{"time": s, "available": s in avail, "label": s} for s in all_slots]
        days_data.append({
            "date": target_date.strftime("%Y-%m-%d"),
            "weekday_short": DAYS_HR[weekday],
            "weekday_full": DAYS_FULL[weekday],
            "is_today": is_today,
            "is_past": is_past,
            "slots": slots,
            "has_available": len(avail) > 0,
        })
    return JsonResponse({
        "service_id": service.id,
        "service_name": service.name,
        "step_minutes": step,
        "duration_min": dur,
        "week_start": week_start.strftime("%Y-%m-%d"),
        "days": days_data,
    })

# ── Slot Availability ───────────────────────────────────────

def get_available_slots(service: "BookingService", target_date: date) -> dict[str, list[str]]:
    """
    Calculate available time slots for a service on a given date.
    Returns dict with 'available' and 'unavailable' slot lists.
    """
    from .models import Appointment, BookingResource, WorkingHours

    if target_date < datetime.now().date():
        return {"available": [], "unavailable": []}

    weekday = target_date.weekday()
    resources = BookingResource.objects.filter(
        resource_type=service.resource_type, is_active=True
    )
    service_duration = service.duration.total_seconds() / 60

    available_slots = set()
    unavailable_slots = set()

    for resource in resources:
        wh = WorkingHours.objects.filter(
            resource=resource, day_of_week=weekday, is_day_off=False
        ).first()
        if not wh:
            continue

        start_dt = datetime.combine(target_date, wh.start_time)
        end_dt = datetime.combine(target_date, wh.end_time)

        existing = list(Appointment.objects.filter(
            resource=resource,
            start_time__date=target_date,
            status__in=["scheduled", "confirmed"],
        ))

        current = start_dt
        while current + timedelta(minutes=service_duration) <= end_dt:
            slot_end = current + timedelta(minutes=service_duration)

            overlap = any(
                current < a.end_time.replace(tzinfo=None)
                and slot_end > a.start_time.replace(tzinfo=None)
                for a in existing
            )

            # Past slots today
            if target_date == datetime.now().date() and current.time() < datetime.now().time():
                overlap = True

            slot_str = current.strftime("%H:%M")
            if not overlap:
                available_slots.add(slot_str)
            else:
                unavailable_slots.add(slot_str)

            current += timedelta(minutes=service.slot_duration_minutes)

    avail = sorted(available_slots)
    unavail = sorted(unavailable_slots - available_slots)
    return {"available": avail, "unavailable": unavail}


def get_week_slots(service: "BookingService", start_date: date, days: int = 7) -> list[dict]:
    """Get available slots for a range of days."""
    result = []
    for i in range(days):
        d = start_date + timedelta(days=i)
        slots = get_available_slots(service, d)
        result.append({
            "date": d.isoformat(),
            "weekday": d.strftime("%A"),
            "available": slots["available"],
            "count": len(slots["available"]),
        })
    return result


# ── Booking Creation ────────────────────────────────────────

@transaction.atomic
def create_booking(service: "BookingService", target_date: date, time_str: str,
                   client_name: str, client_email: str,
                   client_phone: str = "", user: "User | None" = None,
                   company: "Company | None" = None, notes: str = "") -> dict:
    """
    Create a new appointment booking.
    Validates slot availability, creates/finds partner, creates appointment.
    Returns dict with booking details.
    """

    from .models import Appointment

    # Validate slot
    slots = get_available_slots(service, target_date)
    if time_str not in slots["available"]:
        raise BookingConflictError(f"Termin {time_str} nije dostupan za {target_date}.")

    # Parse times
    start_time = datetime.strptime(f"{target_date} {time_str}", "%Y-%m-%d %H:%M")
    duration_min = service.duration.total_seconds() / 60
    end_time = start_time + timedelta(minutes=duration_min)

    # Find available resource
    resource = _find_available_resource(service, target_date, start_time, end_time)
    if not resource:
        raise BookingConflictError("Nema dostupnog resursa za ovaj termin.")

    # Get or create partner
    partner = _get_or_create_partner(
        client_name, client_email, client_phone, company
    )

    # Create appointment
    appointment = Appointment.objects.create(
        service=service,
        resource=resource,
        client=partner,
        start_time=timezone.make_aware(start_time),
        end_time=timezone.make_aware(end_time),
        status="scheduled",
        description=notes,
        company=company,
        employee=user,
    )

    logger.info(
        "Booking created: appt=%d service=%s date=%s time=%s client=%s",
        appointment.pk, service.name, target_date, time_str, client_email,
    )

    return {
        "appointment_id": appointment.pk,
        "service": service.name,
        "date": target_date.isoformat(),
        "time": time_str,
        "duration": int(duration_min),
        "resource": resource.name,
        "client": client_name,
    }


# ── Appointment Status ──────────────────────────────────────

VALID_STATUS_TRANSITIONS = {
    "scheduled": ["confirmed", "cancelled", "no_show"],
    "confirmed": ["completed", "cancelled", "no_show"],
    "completed": [],
    "cancelled": ["scheduled"],
    "no_show": [],
}


def change_appointment_status(appointment: "Appointment", new_status: str, notify: bool = True) -> bool:
    """
    Change appointment status with validation.
    Optionally sends notification to client.
    """
    old = appointment.status
    valid = VALID_STATUS_TRANSITIONS.get(old, [])

    if new_status not in valid:
        raise BizValidationError(f"Nije moguć prijelaz {old} → {new_status}.")

    appointment.status = new_status
    appointment.save(update_fields=["status"])

    if notify and new_status in ("confirmed", "cancelled"):
        _send_status_notification(appointment, new_status)

    logger.info("Appointment %d: %s → %s", appointment.pk, old, new_status)
    return True


# ── Cancellation ────────────────────────────────────────────

def cancel_appointment(appointment: "Appointment", reason: str = "", notify: bool = True) -> bool:
    """Cancel an appointment with optional reason and notification."""
    if appointment.status in ("completed", "no_show"):
        raise BizValidationError("Završeni termini se ne mogu otkazati.")

    appointment.status = "cancelled"
    if reason:
        appointment.description = f"{appointment.description}\nOtkazano: {reason}".strip()
    appointment.save()

    if notify:
        _send_status_notification(appointment, "cancelled")

    logger.info("Appointment %d cancelled: %s", appointment.pk, reason)
    return True


# ── Internal Helpers ────────────────────────────────────────

def _find_available_resource(service: "BookingService", target_date: date, start_time, end_time) -> "BookingResource | None":
    """Find first available resource for service at given time."""
    from .models import Appointment, BookingResource, WorkingHours

    resources = BookingResource.objects.filter(
        resource_type=service.resource_type, is_active=True
    )
    weekday = target_date.weekday()

    for resource in resources:
        wh = WorkingHours.objects.filter(
            resource=resource, day_of_week=weekday, is_day_off=False
        ).first()
        if not wh:
            continue

        # Check working hours
        if start_time.time() < wh.start_time or end_time.time() > wh.end_time:
            continue

        # Check overlaps
        overlap = Appointment.objects.filter(
            resource=resource,
            start_time__date=target_date,
            status__in=["scheduled", "confirmed"],
            start_time__lt=timezone.make_aware(end_time),
            end_time__gt=timezone.make_aware(start_time),
        ).exists()

        if not overlap:
            return resource

    return None


def _get_or_create_partner(name: str, email: str, phone: str = "", company: "Company | None" = None):
    """Get or create partner by email."""
    from partners.models import Partner

    if email:
        partner = Partner.objects.filter(email=email).first()
        if partner:
            return partner

    return Partner.objects.create(
        name=name,
        email=email,
        phone=phone or "",
        company=company,
        is_active=True,
    )


def _send_status_notification(appointment: "Appointment", new_status: str) -> None:
    """Send status change notification (email + WhatsApp) to client."""
    try:
        from .tasks import notify_appointment_status_change
        notify_appointment_status_change.delay(appointment.pk, new_status)
    except Exception as e:
        logger.warning("Failed to send notification for appt %d: %s", appointment.pk, e)

@login_required
def index(request):
    from django.utils import timezone
    today = timezone.now().date()
    # Auto-mark overdue (company-scoped)
    company = get_user_company(request)
    Invoice.objects.filter(
        company=company,
        status__in=['SENT', 'UNPAID'],
        date_due__lt=today

    status_filter  = request.GET.get('status', '')
    country_filter = request.GET.get('country', '')
    doctype_filter = request.GET.get('doctype', '')
    search_q       = request.GET.get('q', '').strip()
    sort_by        = request.GET.get('sort', '-date_issued')
    if status_filter:
        invoices = invoices.filter(status=status_filter)
    if country_filter:
        invoices = invoices.filter(country=country_filter)
    if doctype_filter:
        invoices = invoices.filter(doctype=doctype_filter)
    if search_q:
        from django.db.models import Q
        invoices = invoices.filter(
            Q(number__icontains=search_q) |
            Q(client_name__icontains=search_q) |
            Q(client_oib__icontains=search_q) |
            Q(note__icontains=search_q)
        )
    # Sorting
    allowed_sorts = {
        'date_issued': 'date_issued', '-date_issued': '-date_issued',
        'total': 'total', '-total': '-total',
        'number': 'number', '-number': '-number',
        'client_name': 'client_name', '-client_name': '-client_name',
    }
    sort_field = allowed_sorts.get(sort_by, '-date_issued')
    invoices = invoices.order_by(sort_field)
    from django.db.models import Sum
    all_inv = Invoice.objects.filter(company=company) if company else Invoice.objects.none()
    totals = {
        'total':   all_inv.aggregate(s=Sum('total'))['s'] or 0,
        'unpaid':  all_inv.filter(status__in=['UNPAID','OVERDUE']).aggregate(s=Sum('total'))['s'] or 0,
        'paid':    all_inv.filter(status='PAID').aggregate(s=Sum('total'))['s'] or 0,
        'overdue': all_inv.filter(status='OVERDUE').count(),
        'draft':   all_inv.filter(status='DRAFT').count(),
        'sent':    all_inv.filter(status='SENT').count(),
        'quotes_total': all_inv.filter(doctype='PO').count(),
        'quotes_pending': all_inv.filter(doctype='PO', status__in=['DRAFT', 'SENT']).count(),
    }
    return render(request, 'invoicing/index.html', {
        'invoices': invoices[:200],
        'totals': totals, 'today': today,
        'status_filter': status_filter,
        'country_filter': country_filter,
        'doctype_filter': doctype_filter,
        'search_q': search_q,
        'sort_by': sort_by,
        'status_choices': Invoice.STATUS_CHOICES,
        'doctype_choices': Invoice.DOCTYPE_CHOICES,
    })

        else:

            cert_password = None
            if os.path.exists(cert_password_file):
                with open(cert_password_file) as f_pw:
                    cert_password = f_pw.read().strip()

        result = fiscalize_invoice(
            invoice,
            cert_path=cert_path,
            cert_password=cert_password,
            production=config.use_production
        )

        if result.get('success'):
            invoice.zki = result.get('zki')
            invoice.jir = result.get('jir')
            invoice.is_fiscalized = True
            invoice.fiscal_datetime = timezone.now()
            invoice.operator_oib = invoice.operator.oib if invoice.operator else ''
            if not invoice.qr_code_base64:
                invoice.qr_code_base64 = generate_qr(invoice)
            invoice.save()

            if result.get('demo'):
                messages.success(request, _(f'Račun fiskaliziran (DEMO - CIS nedostupan)!'))
                messages.warning(request, _(f'ZKI: {invoice.zki} | JIR: {invoice.jir}'))
            else:
                messages.success(request, _(f'✓ Račun uspješno fiskaliziran!'))
                messages.success(request, _(f'ZKI: {invoice.zki} | JIR: {invoice.jir}'))
        else:
            messages.error(request, _(f'Greška fiskalizacije: {result.get("error")}'))

    except Exception as e:
        logger.exception("Unexpected error", exc_info=True)
        messages.error(request, _(f'Greška: {e}'))

    return redirect(f'/admin/invoicing/invoice/{pk}/change/')

        issuer_oib=original.issuer_oib,

        issuer_email=original.issuer_email,
        subtotal=-original.subtotal,
        tax_percent=original.tax_percent,
        tax_amount=-original.tax_amount,
        total=-original.total,
        note=f'Storno računa {original.number}',
        storno_of=original,

@transaction.atomic
@login_required
def convert_to_invoice(request, pk):
    """Ponuda ili Predračun → Račun"""
    source = get_object_or_404(Invoice, pk=pk)

    if source.doctype not in ('PO', 'P'):
        messages.error(request, _('Samo ponude ili predračuni se mogu pretvoriti u račun!'))
        return redirect(f'/admin/invoicing/invoice/{pk}/change/')

    if Invoice.objects.unscoped().filter(source_document=source, doctype='R').exists():
        existing = Invoice.objects.unscoped().get(source_document=source, doctype='R')
        messages.warning(request, _(f'Račun već postoji: {existing.number}'))
        return redirect(f'/admin/invoicing/invoice/{existing.pk}/change/')

    from django.utils import timezone

    from .fiscal_models import FiscalConfig
    config = FiscalConfig.get()
    now = timezone.now()

    # Sljedeći broj računa
    last = Invoice.objects.unscoped().filter(doctype='R').order_by('-id').first()
    seq = (last.invoice_sequence + 1) if last else 1
    prefix = config.invoice_prefix_hr if config else 'KNJ'
    new_number = f"{prefix}-{now.year}-{seq:04d}"

    new_doc = _copy_invoice(source, 'R', new_number, {
        'invoice_sequence': seq,
        'date_due': (now + __import__('datetime').timedelta(days=30)).date(),
    })
    new_doc.created_by = request.user
    new_doc.save()

    for item in source.items.all():
        InvoiceItem.objects.create(
            invoice=new_doc,
            description=item.description,
            quantity=item.quantity,
            unit=item.unit,
            unit_price=item.unit_price,
            discount_percent=item.discount_percent,
            total_price=item.total_price,
            tax_rate=item.tax_rate,
            tax_rate_obj=item.tax_rate_obj,
            inventory_item=item.inventory_item,
        )

    source.status = 'ACCEPTED'
    source.save(update_fields=['status'])

    messages.success(request, _(f'Račun {new_doc.number} kreiran iz {source.get_doctype_display()} {source.number}!'))
    return redirect(f'/admin/invoicing/invoice/{new_doc.pk}/change/')

@transaction.atomic
@login_required
def invoice_add_hours(request, pk):
    """GET: return filtered unbilled time entries as JSON.
       POST: create InvoiceItem(s) and mark entries as billed."""
    from decimal import Decimal

    from worktime.models import TimeEntry

    try:
        invoice = Invoice.objects.get(pk=pk)
    except Invoice.DoesNotExist:
        from django.http import Http404
        raise Http404

    if request.method == 'GET':
        qs = TimeEntry.objects.filter(
            status__in=['APPROVED', 'PENDING']
        ).select_related('user', 'project', 'project_lead')

        # filters
        worker_id = request.GET.get('worker', '')
        date_from = request.GET.get('date_from', '')
        date_to   = request.GET.get('date_to', '')
        lead_id   = request.GET.get('lead', '')
        proj_id   = request.GET.get('project', '')
        unbilled_only = request.GET.get('unbilled', '1')

        if worker_id:
            qs = qs.filter(user_id=worker_id)
        if date_from:
            qs = qs.filter(date__gte=date_from)
        if date_to:
            qs = qs.filter(date__lte=date_to)
        if lead_id:
            qs = qs.filter(project_lead_id=lead_id)
        if proj_id:
            qs = qs.filter(project_id=proj_id)
        if unbilled_only == '1':
            qs = qs.filter(is_billed=False)

        qs = qs.order_by('-date')[:200]

        from django.contrib.auth import get_user_model

        from burza.models import ProjectLead
        from worktime.models import Project as WProject

        entries = []
        for e in qs:
            dur = float(e.duration) if e.duration else 0
            billing = float(e.billing_amount) if e.billing_amount else 0
            rate = round(billing / dur, 2) if dur else 0
            entries.append({
                'id': e.pk,
                'date': str(e.date),
                'worker': e.user.get_full_name() or e.user.username,
                'project': e.project.code + ' – ' + e.project.name if e.project else '–',
                'lead': str(e.project_lead) if e.project_lead else '',
                'description': e.description or '',
                'hours': dur,
                'rate': rate,
                'total': billing,
                'is_billed': e.is_billed,
            })

        # dropdown options for filters
        User = get_user_model()
        workers = list(User.objects.filter(
            time_entries__isnull=False
        ).distinct().values('id', 'first_name', 'last_name', 'username'))
        projects = list(WProject.objects.filter(status='ACTIVE').values('id', 'code', 'name'))
        leads = list(ProjectLead.objects.filter(
            status__in=['PREUZETO', 'RIJESENO']
        ).values('id', 'title'))

        return JsonResponse({
            'entries': entries,
            'workers': workers,
            'projects': projects,
            'leads': leads,
        })

    # POST – add to invoice
    import json as _json
    try:
        body = _json.loads(request.body)
    except Exception:
        logger.warning("Suppressed error", exc_info=True)
        body = request.POST.dict()

    mode = body.get('mode', 'ZBIRNO')  # ZBIRNO or POJEDINACNO
    entry_ids = body.get('entry_ids', [])
    tax_rate = Decimal(str(body.get('tax_rate', '25')))

    if not entry_ids:
        return JsonResponse({'error': 'Nema odabranih unosa'}, status=400)

    entries = TimeEntry.objects.filter(pk__in=entry_ids, is_billed=False)
    if not entries.exists():
        return JsonResponse({'error': 'Nema dostupnih unosa'}, status=400)

    if mode == 'ZBIRNO':
        total_hours = sum(float(e.duration or 0) for e in entries)
        total_billing = sum(float(e.billing_amount or 0) for e in entries)
        avg_rate = round(total_billing / total_hours, 2) if total_hours else 0
        dates = sorted(set(str(e.date) for e in entries))
        period = f'{dates[0]} – {dates[-1]}' if len(dates) > 1 else dates[0] if dates else ''
        desc = f'Radni sati {period}'
        item = InvoiceItem(
            invoice=invoice,
            description=desc,
            quantity=Decimal(str(round(total_hours, 2))),
            unit='h',
            unit_price=Decimal(str(avg_rate)),
            tax_rate=tax_rate,
        )
        item.save()
    else:
        for e in entries:
            dur = float(e.duration or 0)
            billing = float(e.billing_amount or 0)
            rate = round(billing / dur, 2) if dur else 0
            worker = e.user.get_full_name() or e.user.username
            proj = e.project.code if e.project else ''
            desc = f'{worker} · {proj} · {e.date}'
            if e.description:
                desc += f' · {e.description[:60]}'
            item = InvoiceItem(
                invoice=invoice,
                description=desc,
                quantity=Decimal(str(round(dur, 2))),
                unit='h',
                unit_price=Decimal(str(rate)),
                tax_rate=tax_rate,
            )
            item.save()

    # Mark entries billed and link to invoice
    entries.update(is_billed=True, invoice=invoice)

    # Recalculate invoice totals
    invoice.recalculate()
    invoice.save()

    return JsonResponse({'ok': True, 'added': len(entry_ids)})

@login_required
def api_field_create(request):
    """POST endpoint for field wizard. Creates invoice from mobile form."""
    if request.method != 'POST':
        return JsonResponse({'success': False, 'error': 'POST required'}, status=405)

    import datetime
    from decimal import Decimal, InvalidOperation

    from .fiscal_models import InvoiceProfile
    from .models import Invoice, InvoiceItem

    try:
        profile = InvoiceProfile.get_default()
        action = request.POST.get('action', 'draft')
        country = request.POST.get('country', 'HR')

        # Create invoice
        invoice = Invoice()
        invoice.doctype = 'R'
        invoice.country = country
        invoice.source = 'wizard_field'
        invoice.date_issued = timezone.now()
        invoice.date_due = (timezone.now() + datetime.timedelta(days=15)).date()

        # Number
        if profile:
            invoice.invoice_profile = profile
            invoice.number = profile.get_next_number('R')
            invoice.issuer_name = profile.company_name
            invoice.issuer_address = f"{profile.address or ''}, {profile.zip_code or ''} {profile.city or ''}".strip(', ')
            invoice.issuer_oib = profile.oib or profile.vat_id or ''
            invoice.issuer_email = profile.email or ''
            invoice.issuer_iban = profile.iban or ''
            invoice.swift = profile.swift or ''
        else:
            from .fiscal_models import FiscalConfig
            config = FiscalConfig.get()
            prefix = config.invoice_prefix_hr or 'KNJ'
            year = datetime.datetime.now().year
            next_num = config.next_invoice_number or 1
            invoice.number = f"{prefix}-{year}-{next_num:04d}"
            while Invoice.objects.filter(number=invoice.number).exists():
                next_num += 1
                invoice.number = f"{prefix}-{year}-{next_num:04d}"
            config.next_invoice_number = next_num + 1
            config.save()

        # Client
        invoice.client_name = request.POST.get('client_name', '')
        invoice.client_address = request.POST.get('client_address', '')
        invoice.client_city = request.POST.get('client_city', '')
        invoice.client_oib = request.POST.get('client_oib', '')
        invoice.client_email = request.POST.get('client_email', '')
        invoice.client_country = 'Deutschland' if country == 'DE' else 'Hrvatska'

        # Partner link
        partner_id = request.POST.get('partner_id')
        if partner_id:
            try:
                from partners.models import Partner
                invoice.partner = Partner.objects.get(pk=partner_id)
            except Exception:
                logger.warning("Suppressed error", exc_info=True)

        # Payment
        invoice.payment_method = 'GOTOVINA' if action == 'paid' else 'TRANSAKCIJSKI'
        invoice.currency = 'EUR'

        # Status based on action
        status_map = {
            'draft': 'DRAFT',
            'issued': 'SENT',
            'email': 'SENT',
            'paid': 'PAID',
        }
        invoice.status = status_map.get(action, 'DRAFT')

        # Notes
        invoice.note = request.POST.get('note', '')
        work_desc = request.POST.get('work_description', '')
        work_cat = request.POST.get('work_category', '')
        if work_desc:
            invoice.internal_note = f"[Teren] {work_cat}: {work_desc}"

        # Photo
        if request.FILES.get('photo'):
            invoice.photo = request.FILES['photo']

        invoice.created_by = request.user
        invoice.save()

        # Items
        descs = request.POST.getlist('item_desc[]')
        qtys = request.POST.getlist('item_qty[]')
        units = request.POST.getlist('item_unit[]')
        prices = request.POST.getlist('item_price[]')
        pdvs = request.POST.getlist('item_pdv[]')

        for i, desc in enumerate(descs):
            if not desc.strip():
                continue
            try:
                qty = Decimal(str(qtys[i]).replace(',', '.')) if i < len(qtys) else Decimal('1')
                price = Decimal(str(prices[i]).replace(',', '.')) if i < len(prices) else Decimal('0')
                pdv = Decimal(str(pdvs[i])) if i < len(pdvs) else Decimal('25')
                unit = units[i] if i < len(units) else 'kom'
                disc = Decimal('0')
                total = qty * price * (1 - disc / 100)
                InvoiceItem.objects.create(
                    invoice=invoice,
                    description=desc,
                    quantity=qty,
                    unit=unit,
                    unit_price=price,
                    tax_rate=pdv,
                    discount_percent=disc,
                    total_price=round(total, 2),
                )
            except (InvalidOperation, IndexError, ValueError):
                continue

        invoice.recalculate()
        invoice.save()

        # Send email if requested
        if action == 'email' and invoice.client_email:
            try:
                from core.email_tasks import send_email_with_attachment_task
                # Generate PDF first, then email (async)
                send_email_with_attachment_task.delay(invoice.pk)
            except Exception:
                logger.warning("Suppressed error", exc_info=True)

        return JsonResponse({
            'success': True,
            'invoice_id': invoice.pk,
            'number': invoice.number,
            'total': float(invoice.total),
        })

    except Exception as e:
        logger.exception("Unexpected error", exc_info=True)
        import traceback
        traceback.print_exc()
        return JsonResponse({'success': False, 'error': str(e)}, status=500)

def api_update_item(request, pk):
    """PATCH endpoint to update a single InvoiceItem inline."""
    if request.method != 'PATCH':
        return JsonResponse({'success': False, 'error': 'PATCH required'}, status=405)

    import json
    from decimal import Decimal

    from .models import InvoiceItem

    try:
        item = InvoiceItem.objects.select_related('invoice').get(pk=pk)
    except InvoiceItem.DoesNotExist:
        return JsonResponse({'success': False, 'error': 'Item not found'}, status=404)

    try:
        data = json.loads(request.body)
    except json.JSONDecodeError:
        return JsonResponse({'success': False, 'error': 'Invalid JSON'}, status=400)

    # Update only provided fields
    if 'description' in data:
        item.description = data['description']
    if 'quantity' in data:
        item.quantity = Decimal(str(data['quantity']).replace(',', '.'))
    if 'unit' in data:
        item.unit = data['unit']
    if 'unit_price' in data:
        item.unit_price = Decimal(str(data['unit_price']).replace(',', '.'))
    if 'discount_percent' in data:
        item.discount_percent = Decimal(str(data['discount_percent']).replace(',', '.'))
    if 'tax_rate' in data:
        item.tax_rate = Decimal(str(data['tax_rate']))

    item.save()  # auto-calculates total_price and gross_total

    # Recalculate invoice totals
    invoice = item.invoice
    invoice.recalculate()
    invoice.save()

    return JsonResponse({
        'success': True,
        'item': {
            'id': item.pk,
            'total_price': float(item.total_price),
            'gross_total': float(item.gross_total),
        },
        'invoice': {
            'subtotal': float(invoice.subtotal),
            'tax_amount': float(invoice.tax_amount),
            'total': float(invoice.total),
        },
    })

    """PATCH endpoint to update invoice header fields."""
    if request.method != 'PATCH':
        return JsonResponse({'success': False, 'error': 'PATCH required'}, status=405)

    import json

    from .models import Invoice

    try:
        invoice = Invoice.objects.get(pk=pk)
    except Invoice.DoesNotExist:
        return JsonResponse({'success': False, 'error': 'Invoice not found'}, status=404)

    try:
        data = json.loads(request.body)
    except json.JSONDecodeError:
        return JsonResponse({'success': False, 'error': 'Invalid JSON'}, status=400)

    allowed_fields = [
        'client_name', 'client_address', 'client_city', 'client_oib', 'client_email',
        'note', 'internal_note', 'payment_method', 'reference_number',
        'date_due', 'date_delivery',
    ]

    import datetime
    for field in allowed_fields:
        if field in data:
            val = data[field]
            if field in ('date_due', 'date_delivery') and val:
                try:
                    val = datetime.date.fromisoformat(val)
                except (ValueError, TypeError):
                    continue
            setattr(invoice, field, val)

    invoice.save()
    return JsonResponse({'success': True, 'updated': list(data.keys())})

def fiscal_batch_create(request):
    """Batch fiskalizacija odabranih računa."""
    from django.utils import timezone as tz
    from .fiscal_models import FiscalBatch
    company = get_user_company(request)
    if not company:
        messages.error(request, 'Nije pronađena tvrtka.')
        return redirect('invoicing:fiscal_batch')
    invoice_ids = request.POST.getlist('invoice_ids')
    if not invoice_ids:
        messages.warning(request, 'Niste odabrali nijedan račun.')
        return redirect('invoicing:fiscal_batch')
    invoices = Invoice.objects.filter(
        pk__in=invoice_ids, company=company, is_fiscalized=False,
    )
    if not invoices.exists():
        messages.warning(request, 'Nema odabranih nefiskaliziranih računa.')
        return redirect('invoicing:fiscal_batch')
    batch = FiscalBatch.objects.create(
        company=company,
        created_by=request.user,
        status='processing',
        total_count=invoices.count(),
        started_at=tz.now(),
    )
    batch.invoices.set(invoices)
    success, errors = 0, 0
    results = {}
    try:
        from fiscal.service import fiscalize_invoice
        from .fiscal_models import FiscalConfig
        config = FiscalConfig.get()
        cert_path = config.cert_file.path if config.cert_file else config.cert_path
        cert_password = config.cert_password or None
    except Exception:
        fiscalize_invoice = None
        cert_path = cert_password = None
        config = None
    for inv in invoices:
        try:
            if fiscalize_invoice and cert_path:
                result = fiscalize_invoice(inv, cert_path=cert_path,
                                           cert_password=cert_password,
                                           production=getattr(config, 'use_production', False))
            else:
                result = {'success': False, 'error': 'Fiskalizacija nije konfigurirana'}
            if result.get('success'):
                inv.zki = result.get('zki', '')
                inv.jir = result.get('jir', '')
                inv.is_fiscalized = True
                inv.fiscal_datetime = tz.now()
                inv.save(update_fields=['zki', 'jir', 'is_fiscalized', 'fiscal_datetime'])
                results[inv.pk] = {'ok': True}
                success += 1
            else:
                results[inv.pk] = {'ok': False, 'error': result.get('error', 'Nepoznata greška')}
                errors += 1
        except Exception as exc:
            results[inv.pk] = {'ok': False, 'error': str(exc)}
            errors += 1
    batch.success_count = success
    batch.error_count = errors
    batch.results_json = results
    batch.status = 'completed' if errors == 0 else ('partial' if success > 0 else 'failed')
    batch.completed_at = tz.now()
    batch.save()
    messages.success(request, f'Batch završen: {success} uspješno, {errors} grešaka.')
    return redirect('invoicing:fiscal_batch')

@login_required
def work_order_to_invoice(request, pk):
    """Create invoice from work order and redirect to invoice edit."""
    company = _get_company(request)
    wo = get_object_or_404(WorkOrder, pk=pk, company=company)

    if wo.status == 'invoiced':
        messages.warning(request, "Nalog je već fakturiran.")
        return redirect('work_orders:detail', pk=pk)

    if request.method == 'POST':
        with transaction.atomic():
            from invoicing.models import Invoice, InvoiceItem
            import datetime

            # Generate invoice number same way as invoicing app
            last = Invoice.objects.order_by('-id').first()
            now = datetime.date.today()
            next_num = f"KNJ-{now.year}-{(last.id + 1 if last else 1):04d}"

            client_name = wo.client.name if wo.client else ''
            client_address = ''
            client_oib = ''
            if wo.client:
                client_address = getattr(wo.client, 'address', '') or ''
                client_oib = getattr(wo.client, 'oib', '') or ''

            invoice = Invoice.objects.create(
                company=company,
                doctype='R',
                number=next_num,
                status='DRAFT',
                date_issued=datetime.datetime.now(),
                date_due=now + datetime.timedelta(days=30),
                client_name=client_name,
                client_address=client_address,
                client_oib=client_oib,
                note=f"Radni nalog {wo.order_number}\n{wo.notes}".strip(),
                created_by=request.user,
            )

            for item in wo.items.select_related('article').all():
                InvoiceItem.objects.create(
                    invoice=invoice,
                    description=item.description,
                    quantity=item.quantity,
                    unit=item.unit,
                    unit_price=item.unit_price,
                    discount_percent=item.discount_percent,
                    tax_rate=item.tax_rate,
                    inventory_item=item.article,
                )

            # Link work order to invoice
            wo.invoice = invoice
            wo.status = 'invoiced'
            wo.save(update_fields=['invoice', 'status'])

        messages.success(request, f"Račun kreiran iz naloga {wo.order_number}.")
        return redirect('invoicing:invoice_edit', pk=invoice.pk)

    return render(request, 'work_orders/work_order_to_invoice.html', {'wo': wo})

def _save_items(wo, post):
    """Parse POST arrays and create WorkOrderItems."""
    item_types = post.getlist('item_type[]')
    descriptions = post.getlist('item_description[]')
    quantities = post.getlist('item_quantity[]')
    units = post.getlist('item_unit[]')
    prices = post.getlist('item_price[]')
    discounts = post.getlist('item_discount[]')
    tax_rates = post.getlist('item_tax_rate[]')
    article_ids = post.getlist('item_article_id[]')

    for i, desc in enumerate(descriptions):
        if not desc.strip():
            continue
        try:
            qty = float(quantities[i]) if i < len(quantities) else 1
            price = float(prices[i]) if i < len(prices) else 0
            disc = float(discounts[i]) if i < len(discounts) else 0
            tax = float(tax_rates[i]) if i < len(tax_rates) else 25
            itype = item_types[i] if i < len(item_types) else 'labor'
            unit = units[i] if i < len(units) else 'sat'
            art_id = article_ids[i] if i < len(article_ids) else ''
        except (ValueError, TypeError):
            continue

        item = WorkOrderItem(
            work_order=wo,
            item_type=itype,
            description=desc.strip(),
            quantity=qty,
            unit=unit,
            unit_price=price,
            discount_percent=disc,
            tax_rate=tax,
        )
        if art_id:
            try:
                item.article_id = int(art_id)
            except (ValueError, TypeError):
                pass
        item.save()


def _fill_client_card(cc, post):
    """Fill ClientCard fields from POST data."""
    def _dec(key):
        v = post.get(key, '').strip()
        from decimal import Decimal, InvalidOperation
        try:
            return Decimal(v) if v else None
        except InvalidOperation:
            return None

    def _int(key):
        v = post.get(key, '').strip()
        try:
            return int(v) if v else None
        except (ValueError, TypeError):
            return None

    # Optics
    cc.sphere_right = _dec('sphere_right')
    cc.cylinder_right = _dec('cylinder_right')
    cc.axis_right = _int('axis_right')
    cc.sphere_left = _dec('sphere_left')
    cc.cylinder_left = _dec('cylinder_left')
    cc.axis_left = _int('axis_left')
    cc.pd = _dec('pd')
    cc.add_power = _dec('add_power')
    # Hair
    cc.hair_type = post.get('hair_type', '')
    cc.allergies = post.get('allergies', '')
    # General
    cc.gdpr_consent = 'gdpr_consent' in post
    last_visit = post.get('last_visit', '').strip()
    cc.last_visit = last_visit or None
    cc.notes = post.get('notes', '')

@login_required
def partner_list(request):
    q = request.GET.get("q", "")
    ptype = request.GET.get("type", "")
    active = request.GET.get("active", "")
    company = get_user_company(request)
    qs = Partner.objects.filter(company=company).order_by("name") if company else Partner.objects.none()
    if q:
        qs = qs.filter(Q(name__icontains=q)|Q(email__icontains=q)|Q(oib__icontains=q)|Q(city__icontains=q))
    if ptype:
        qs = qs.filter(partner_type=ptype)
    if active == "1":
        qs = qs.filter(is_active=True)
    elif active == "0":
        qs = qs.filter(is_active=False)
    elif active == "incomplete":
        qs = qs.filter(Q(oib="")|Q(address="")|Q(city="")|Q(email=""))
    paginator = Paginator(qs, 30)
    page = paginator.get_page(request.GET.get("page"))
    return render(request, "partners/partner_list.html", {
        "page_obj": page, "q": q, "ptype": ptype, "active": active,
        "total": qs.count(), "type_choices": Partner.TYPE_CHOICES,
    })


@login_required

@login_required
def partner_create(request):
    companies = [get_user_company(request)] if get_user_company(request) else []
    if request.method == "POST":
        try:
            company = get_user_company(request)
            if not company:
                messages.error(request, 'Nema aktivne firme.')
                return redirect('dashboard')
            p = Partner(
                company=company,
                name=request.POST["name"],
                oib=request.POST.get("oib", "").strip() or None,
                address=request.POST.get("address", ""),
                city=request.POST.get("city", ""),
                zip_code=request.POST.get("zip_code", ""),
                country=request.POST.get("country", "Hrvatska"),
                email=request.POST.get("email", ""),
                phone=request.POST.get("phone", ""),
                web=request.POST.get("web", ""),
                partner_type=request.POST.get("partner_type", "client"),
                payment_term=int(request.POST.get("payment_term", 0) or 0),
                discount_percent=request.POST.get("discount_percent", 0) or 0,
                notes=request.POST.get("notes", ""),
            )
            nkd_id = request.POST.get("nkd_code_id", "").strip()
            if nkd_id:
                try:
                    p.nkd_code_id = int(nkd_id)
                except (ValueError, TypeError):
                    pass
            p.save()
            messages.success(request, f"Partner {p.name} kreiran.")
            return redirect("partners:detail", pk=p.pk)
        except Exception as e:
            messages.error(request, str(e))
    return render(request, "partners/partner_form.html", {
        "companies": companies, "type_choices": Partner.TYPE_CHOICES,
        "action": "Novi partner",
    })

    companies = [get_user_company(request)] if get_user_company(request) else []
    if request.method == "POST":
        try:
            # Company remains unchanged on edit — only admins may reassign via admin panel
            partner.name = request.POST["name"]
            partner.oib = request.POST.get("oib", "").strip() or None
            partner.address = request.POST.get("address", "")
            partner.city = request.POST.get("city", "")
            partner.zip_code = request.POST.get("zip_code", "")
            partner.country = request.POST.get("country", "Hrvatska")
            partner.email = request.POST.get("email", "")
            partner.phone = request.POST.get("phone", "")
            partner.web = request.POST.get("web", "")
            partner.partner_type = request.POST.get("partner_type", "client")
            partner.payment_term = int(request.POST.get("payment_term", 0) or 0)
            partner.discount_percent = request.POST.get("discount_percent", 0) or 0
            partner.notes = request.POST.get("notes", "")
            partner.is_active = "is_active" in request.POST

def partner_quick_create(request):
    import json
    try:
        data = json.loads(request.body)
    except Exception:
        data = request.POST

    name = (data.get('name') or '').strip()
    oib  = (data.get('oib') or '').strip()

    if not name:
        return JsonResponse({'ok': False, 'error': 'Naziv je obavezan.'}, status=400)

    company = get_user_company(request)
    if not company:
        return JsonResponse({'ok': False, 'error': 'Nema aktivne firme.'}, status=403)

    # Duplicate check by OIB first, then name
    existing = None
    if oib:
        existing = Partner.objects.filter(company=company, oib=oib).first()
    if not existing:
        existing = Partner.objects.filter(company=company, name__iexact=name).first()

    if existing:
        return JsonResponse({
            'ok': True, 'status': 'exists',
            'id': existing.pk, 'name': existing.name,
            'oib': existing.oib, 'email': existing.email,
            'address': existing.address, 'city': existing.city,
        })

    p = Partner(
        company=company,
        name=name,
        oib=oib,
        email=(data.get('email') or '').strip(),
        phone=(data.get('phone') or '').strip(),
        address=(data.get('address') or '').strip(),
        city=(data.get('city') or '').strip(),
        partner_type=(data.get('partner_type') or 'client'),
    )
    p.save()
    return JsonResponse({
        'ok': True, 'status': 'created',
        'id': p.pk, 'name': p.name,
        'oib': p.oib, 'email': p.email,
        'address': p.address, 'city': p.city,
    }, status=201)



@login_required
def partner_oib_lookup(request):
    """AJAX: Dohvati podatke firme po OIB-u / VAT ID-u."""
    oib = request.GET.get('oib', '').strip()
    country = request.GET.get('country', 'HR').strip().upper()
    if len(country) > 3:
        country = 'HR'

    if not oib:
        return JsonResponse({'error': 'OIB je obavezan'}, status=400)

    from core.oib_validator import validate_vat_id
    if not validate_vat_id(oib, country):
        return JsonResponse({'error': 'Neispravan OIB/VAT format', 'valid_format': False}, status=400)

    company = get_user_company(request)

    # Provjeri postoji li u bazi
    if company:
        existing = Partner.objects.filter(company=company, oib=oib).first()
        if existing:
            return JsonResponse({
                'found': True,
                'source': 'database',
                'name': existing.name,
                'oib': existing.oib,
                'address': existing.address or '',
                'city': existing.city or '',
                'email': existing.email or '',
                'country': existing.country or country,
                'partner_id': existing.pk,
            })

    # Dohvati iz VIES registra
    try:
        from .hr_registry import PartnerAutoCreator
        creator = PartnerAutoCreator()
        result = creator.lookup_oib(oib, country)
    except Exception as exc:
        return JsonResponse({'found': False, 'error': 'Greska pri dohvatu: ' + str(exc)})

    if result.get('valid'):
        return JsonResponse({
            'found': True,
            'source': 'vies',
            'name': result.get('name', ''),
            'oib': oib,
            'address': result.get('address', ''),
            'city': result.get('city', ''),
            'zip_code': result.get('zip_code', ''),
            'email': '',
            'country': country,
        })

    return JsonResponse({
        'found': False,
        'source': 'vies',
        'error': result.get('error') or 'OIB nije pronaden u registru',
    })

def _require_company_admin(request):
    try:
        profile = request.user.profile
    except Exception:
        return None, None
    if not profile.is_company_admin:
        return None, None
    return profile, profile.company


@login_required
def company_settings(request):
    profile, company = _require_company_admin(request)
    if not company:
        messages.error(request, "Nemate pristup postavkama tvrtke.")
        return redirect("/")

    tab = request.GET.get("tab", "info")

    # Tab: info
    if tab == "info" and request.method == "POST":
        company.name = request.POST.get("name", company.name)
        company.oib = request.POST.get("oib", company.oib)
        company.address = request.POST.get("address", "")
        company.city = request.POST.get("city", "")
        company.zip_code = request.POST.get("zip_code", "")
        company.country = request.POST.get("country", company.country)
        company.iban = request.POST.get("iban", "")
        company.phone = request.POST.get("phone", "")
        company.email = request.POST.get("company_email", "")
        company.website_url = request.POST.get("website_url", "")
        company.mobile = request.POST.get("mobile", "")
        company.tax_number = request.POST.get("tax_number", "")
        company.vat_id = request.POST.get("vat_id", "")
        company.business_type = request.POST.get("business_type", company.business_type)
        company.currency = request.POST.get("currency", company.currency)
        company.fiscal_system = request.POST.get("fiscal_system", company.fiscal_system)
        company.pdv_status = request.POST.get("pdv_status", company.pdv_status)
        company.is_vat_registered = bool(request.POST.get("is_vat_registered"))
        if "logo" in request.FILES:
            company.logo = request.FILES["logo"]
        nkd_id = request.POST.get("nkd_code_id", "").strip()
        if nkd_id:
            try:
                from .models import NKDCode
                company.nkd_code = NKDCode.objects.get(pk=nkd_id)
            except Exception:
                pass
        elif "nkd_code_id" in request.POST:
            company.nkd_code = None
        company.save()
        messages.success(request, "Podaci tvrtke su azurirani.")
        return redirect("/companies/settings/?tab=info")

    # Tab: settings
    if tab == "settings" and request.method == "POST":
        # Company model fields
        company.business_type = request.POST.get("business_type", company.business_type)
        company.currency = request.POST.get("currency", company.currency)
        # JSON settings
        s = company.settings or {}
        s["default_tax_pct"] = request.POST.get("default_tax_pct", "25")
        s["invoice_number_format"] = request.POST.get("invoice_number_format", "{YYYY}-{NUM}")
        s["offer_number_format"] = request.POST.get("offer_number_format", "PON-{YYYY}-{NUM}")
        s["email_reply_to"] = request.POST.get("email_reply_to", "")
        s["working_hours"] = request.POST.get("working_hours", "08:00-16:00")
        s["ui_language"] = request.POST.get("ui_language", "hr")
        s["send_invoice_email"] = request.POST.get("send_invoice_email") == "1"
        s["send_offer_email"] = request.POST.get("send_offer_email") == "1"
        company.settings = s
        company.save()
        messages.success(request, "Postavke spremljene.")
        return redirect("/companies/settings/?tab=settings")

    # Tab: fiscal
    if tab == "fiscal" and request.method == "POST":
        company.fiscal_system = request.POST.get("fiscal_system", company.fiscal_system)
        company.pdv_status = request.POST.get("pdv_status", company.pdv_status)
        company.is_vat_registered = request.POST.get("is_vat_registered") == "1"
        company.is_kleinunternehmer = request.POST.get("is_kleinunternehmer") == "1"
        company.vat_id = request.POST.get("vat_id", company.vat_id)
        company.save(update_fields=[
            "fiscal_system", "pdv_status", "is_vat_registered",
            "is_kleinunternehmer", "vat_id",
        ])
        messages.success(request, "Fiskalne postavke su azurirane.")
        return redirect("/companies/settings/?tab=fiscal")

    # Context
    employees = UserProfile.objects.filter(company=company).select_related("user")
    branches = company.branches.all()

    sub = company.subscriptions.filter(is_active=True).first()
    usage = {
        "users": employees.count(),
        "locations": branches.filter(is_active=True).count(),
        "invoices": 0,
    }
    try:
        from invoicing.models import Invoice
        usage["invoices"] = Invoice.objects.filter(company=company).count()
    except Exception:
        pass

    # Fiscal certificates for fiscal tab
    fiscal_certs = []
    try:
        from django.utils import timezone as _tz

        from invoicing.fiscal_models import FiscalCertificate
        fiscal_certs = list(
            FiscalCertificate.objects.filter(company=company).select_related('location').order_by('-uploaded_at')
        )
        _now = _tz.now()
        for c in fiscal_certs:
            c.is_expired = c.valid_until and c.valid_until < _now
    except Exception:
        pass

    return render(request, "companies/settings.html", {
        "company": company,
        "tab": tab,
        "employees": employees,
        "branches": branches,
        "subscription": sub,
        "usage": usage,
        "fiscal_certs": fiscal_certs,
        "role_choices": UserProfile.ROLE_CHOICES,
        "branch_types": CompanyBranch.BRANCH_TYPES,
        "country_choices": Company.Country.choices,
        "settings_data": company.settings or {},
    })



@login_required
@require_POST
def employee_create(request):
    profile, company = _require_company_admin(request)
    if not company:
        return JsonResponse({"error": "Forbidden"}, status=403)

    email = request.POST.get("email", "").strip()
    first_name = request.POST.get("first_name", "").strip()
    last_name = request.POST.get("last_name", "").strip()
    role = request.POST.get("role", "worker")
    phone = request.POST.get("phone", "").strip()

    if not email:
        return JsonResponse({"error": "Email je obavezan."}, status=400)

    from django.contrib.auth.models import User
    if User.objects.filter(email=email).exists():
        return JsonResponse({"error": "Korisnik s tim emailom vec postoji."}, status=400)

    alphabet = string.ascii_letters + string.digits
    temp_pw = "".join(secrets.choice(alphabet) for _ in range(12))

    base_un = email.split("@")[0].lower().replace(".", "_")[:30]
    username = base_un
    counter = 1
    while User.objects.filter(username=username).exists():
        username = base_un + str(counter)
        counter += 1

    user = User.objects.create_user(
        username=username, email=email, password=temp_pw,
        first_name=first_name, last_name=last_name,
    )
    if role in ("company_admin", "project_manager"):
        user.is_staff = True
        user.save()

    up, created = UserProfile.objects.get_or_create(user=user, defaults={
        "company": company, "role": role, "phone": phone,
        "must_change_password": True,
    })
    if not created:
        up.company = company
        up.role = role
        up.phone = phone
        up.must_change_password = True
        up.save()

    try:
        from django.core.mail import send_mail
        send_mail(
            "Vas pristup - " + company.name,
            "Postovani " + first_name + ",\n\nKreiran je korisnicki racun za LIMIT Connect.\n\n"
            + "Email: " + email + "\nLozinka: " + temp_pw + "\n\n"
            + "Prijavite se na: https://limit-connect.com/login/\n"
            + "Prilikom prve prijave bit cete zamoljeni da promijenite lozinku.\n\n"
            + "Pozdrav,\n" + company.name,
            "noreply@limit-connect.com",
            [email],
            fail_silently=True,
        )
    except Exception:
        pass

    return JsonResponse({"ok": True, "message": "Zaposlenik " + first_name + " " + last_name + " kreiran.", "temp_password": temp_pw})

def employee_change_password(request, pk):
    profile, company = _require_company_admin(request)
    if not company:
        return JsonResponse({"error": "Forbidden"}, status=403)

    up = get_object_or_404(UserProfile, pk=pk, company=company)
    new_pw = request.POST.get("password", "").strip()
    if len(new_pw) < 8:
        return JsonResponse({"error": "Lozinka mora imati najmanje 8 znakova."}, status=400)
    up.user.set_password(new_pw)
    up.user.save()
    up.must_change_password = True
    up.save(update_fields=["must_change_password"])
    return JsonResponse({"ok": True, "message": "Lozinka promijenjena."})

@login_required
def nkd_search(request):
    from django.db.models import Q
    from django.http import JsonResponse

    from .models import NKDCode
    q = request.GET.get("q", "").strip()
    level = request.GET.get("level", "")
    if not q or len(q) < 2:
        return JsonResponse([], safe=False)
    qs = NKDCode.objects.filter(is_active=True).filter(
        Q(name_hr__icontains=q) | Q(code__icontains=q) | Q(name_en__icontains=q)
    )
    if level:
        qs = qs.filter(level=level)
    results = list(qs.values("id", "code", "level", "name_hr", "name_en", "section")[:20])
    return JsonResponse(results, safe=False)

def branch_save_redirect(request, pk=None):
    """Wrapper around branch_save that redirects instead of returning JSON."""
    profile, company = _require_company_admin(request)
    if not company:
        messages.error(request, 'Nemate pristup.')
        return redirect('/companies/settings/?tab=branches')

    if pk:
        from .models import CompanyBranch
        branch = get_object_or_404(CompanyBranch, pk=pk, parent_company=company)
    else:
        from .models import CompanyBranch
        branch = CompanyBranch(parent_company=company)

    branch.name = request.POST.get('name', '').strip()
    branch.branch_code = request.POST.get('branch_code', '').strip()
    branch.branch_type = request.POST.get('branch_type', 'other')
    branch.address = request.POST.get('address', '').strip()
    branch.city = request.POST.get('city', '').strip()
    branch.phone = request.POST.get('phone', '').strip()
    branch.email = request.POST.get('email', '').strip()
    try:
        branch.save()
        messages.success(request, f'Podružnica {branch.name} spremljena.')
    except Exception as e:
        messages.error(request, f'Greška: {e}')
    return redirect('/companies/settings/?tab=branches')

def dashboard_router(request):
    """Preusmjeri ovisno o domeni i statusu prijave"""
    host = request.get_host().split(':')[0]
    # Ako je viskovic domena
    if 'viskovic' in host:
        if request.user.is_authenticated:
            return redirect('https://limit-connect.com/dashboard/')
        from website.views import homepage
        return homepage(request)
    # limit-connect.com - nelogirani vide landing page
    if not request.user.is_authenticated:
        return render(request, 'landing.html')
    # Ako nije prijavljen - na login
    if not request.user.is_authenticated:
        return redirect('login')

    # Ako je prijavljen - na dashboard prema roli
    if request.user.is_superuser:
        # Ako je superadmin u impersonation modu -> company_dashboard
        if request.session.get('impersonated_company_id') or request.session.get('superadmin_as_company'):
            return redirect('company_dashboard')
        return redirect('superadmin_dashboard')

    # ── Branch routing: branch users → branch-specific dashboard ──────────
    try:
        branch = request.user.profile.branch
        if branch and branch.is_active:
            return redirect(branch.dashboard_url)
    except Exception:
        logger.warning("Suppressed error", exc_info=True)

    # Route by role
    try:
        role = request.user.profile.role
    except Exception:
        role = 'worker'

    if role in ('company_admin', 'project_manager'):
        return redirect('company_dashboard')
    elif role == 'investor' or role == 'partner':
        return redirect('partner_dashboard') # Šaljemo ih oboje u Staklenu sobu
    else:
        # worker / unknown → worker dashboard
        return redirect('worker_dashboard')

def login_view(request):
    if request.user.is_authenticated:
        return redirect('dashboard')

    if request.method == 'POST':
        username = request.POST.get('username')
        password = request.POST.get('password')
        user = authenticate(request, username=username, password=password)
        if user:
            login(request, user)
            # Ako se logira na viskovic domeni, redirectaj na ERP
            host = request.get_host().split(':')[0]
            if 'viskovic' in host:
                return redirect('https://limit-connect.com/dashboard/')
            next_url = request.GET.get('next', '/')
            return redirect(next_url)
        else:
            return render(request, 'registration/login.html', {'error': 'Ungültige Anmeldedaten'})

    return render(request, 'registration/login.html')

def logout_view(request):
    logout(request)
    host = request.get_host().split(':')[0]
    if 'viskovic' in host:
        return redirect('/')
    return redirect('login')

@login_required
def company_dashboard(request):
    import datetime

    from django.db.models import Sum
    from django.utils import timezone

    company = None
    try:
        company = request.user.profile.company
    except Exception:
        logger.warning("Suppressed error", exc_info=True)

    today = timezone.now().date()
    week_start = today - datetime.timedelta(days=today.weekday())
    month_start = today.replace(day=1)

    # ── Fakture ──────────────────────────────────────────────────────────────
    try:
        from invoicing.models import Invoice
        inv_qs = Invoice.objects.unscoped()
        if company:
            inv_qs = inv_qs.filter(company=company)

        total_invoiced = inv_qs.filter(
            doctype='R', status__in=('SENT', 'PAID', 'UNPAID', 'OVERDUE')
        ).aggregate(s=Sum('total'))['s'] or 0

        unpaid = inv_qs.filter(
            doctype='R', status__in=('UNPAID', 'OVERDUE')
        ).aggregate(s=Sum('total'))['s'] or 0

        paid_month = inv_qs.filter(
            doctype='R', status='PAID',
            date_issued__date__gte=month_start
        ).aggregate(s=Sum('total'))['s'] or 0

        recent_invoices = inv_qs.filter(doctype='R').order_by('-date_issued')[:8]

        pending_offers = inv_qs.filter(
            doctype='PO', status__in=('DRAFT', 'SENT')
        ).count()
    except Exception:
        total_invoiced = unpaid = paid_month = pending_offers = 0
        recent_invoices = []

    # ── Radni sati ────────────────────────────────────────────────────────────
    try:
        from worktime.models import TimeEntry
        te_qs = TimeEntry.objects.filter(user=request.user)
        hours_week  = te_qs.filter(date__gte=week_start).aggregate(s=Sum('duration'))['s'] or 0
        hours_month = te_qs.filter(date__gte=month_start).aggregate(s=Sum('duration'))['s'] or 0
        recent_entries = te_qs.select_related('project').order_by('-date')[:5]
    except Exception:
        hours_week = hours_month = 0
        recent_entries = []

    # ── Projekti ──────────────────────────────────────────────────────────────
    try:
        from projects.models import Project
        proj_qs = Project.objects.unscoped()
        if company:
            proj_qs = proj_qs.filter(company=company)
        projects = proj_qs.filter(status__in=('active', 'in_progress', 'open')).order_by('-id')[:5]
        total_projects = proj_qs.exclude(status__in=('completed', 'cancelled')).count()
    except Exception:
        projects = []
        total_projects = 0

    # ── Partneri ──────────────────────────────────────────────────────────────
    try:
        from partners.models import Partner
        part_qs = Partner.objects.unscoped()
        if company:
            part_qs = part_qs.filter(company=company)
        total_partners = part_qs.count()
    except Exception:
        total_partners = 0

    try:
        from finance.models import Transaction
        tx_qs = Transaction.objects.unscoped()
        if company:
            tx_qs = tx_qs.filter(company=company)

def health_check(request):
    """Endpoint za monitoring – vraća JSON status svih servisa"""
    from django.db import connection
    from django.http import JsonResponse
    status = {"status": "ok", "services": {}}
    # DB
    try:
        with connection.cursor() as c:
            c.execute("SELECT 1")
        status["services"]["database"] = "ok"
    except Exception as e:
        status["services"]["database"] = f"error: {e}"
        status["status"] = "degraded"
    # Cache
    try:
        from django.core.cache import cache
        cache.set("health_check", "1", 10)
        val = cache.get("health_check")
        status["services"]["cache"] = "ok" if val == "1" else "miss"
    except Exception as e:
        status["services"]["cache"] = f"error: {e}"
    # Gunicorn workers
    try:
        import psutil
        workers = [p for p in psutil.process_iter(["name", "cmdline"])
                   if "gunicorn" in " ".join(p.info.get("cmdline", []))]
        status["services"]["workers"] = len(workers)
    except Exception:
        status["services"]["workers"] = "unknown"
    return JsonResponse(status)

    """Superadmin: toggle individual modules per company subscription."""

        from django.http import HttpResponseForbidden
        return HttpResponseForbidden()

def start_impersonation(request, company_id):
    """Superadmin begins impersonating a company."""
    if not request.user.is_superuser:
        from django.http import HttpResponseForbidden
        return HttpResponseForbidden()
    import time

    from django.shortcuts import get_object_or_404

    from accounts.models import ImpersonationLog
    from companies.models import Company

    company = get_object_or_404(Company, id=company_id)

    # Resolve real IP
    x_fwd = request.META.get('HTTP_X_FORWARDED_FOR')
    ip = x_fwd.split(',')[0].strip() if x_fwd else request.META.get('REMOTE_ADDR')

    # Close any existing session first
    old_log = request.session.get('impersonation_log_id')
    if old_log:
        from django.utils import timezone
        ImpersonationLog.objects.filter(id=old_log).update(ended_at=timezone.now())

    log = ImpersonationLog.objects.create(
        superadmin=request.user,
        company=company,
        ip_address=ip,
    )
    request.session['impersonated_company_id'] = str(company.id)
    request.session['impersonation_started_at'] = int(time.time())
    request.session['impersonation_log_id'] = log.id

    messages.success(request, f'Pristupate kao tvrtka: {company.name}')
    return redirect('company_dashboard')


@login_required
def stop_impersonation(request):
    """Superadmin ends the impersonation session."""
    from django.utils import timezone
    log_id = request.session.get('impersonation_log_id')
    if log_id:
        from accounts.models import ImpersonationLog

@require_role("superadmin")
@login_required
def superadmin_companies(request):
    """List all companies; create new ones; change plans; start impersonation."""
    if not request.user.is_superuser:
        return redirect('dashboard')

    import datetime as _dt
    from decimal import Decimal

    from django.contrib.auth import get_user_model
    from django.db.models import Max

    from companies.models import Company
    from registration.constants import PLAN_PRICES
    from registration.models import CompanyRegistration, CompanySubscription
    User = get_user_model()

    error = None

    # ── Inline plan change ────────────────────────────────────────
    if request.method == 'POST' and request.POST.get('action') == 'change_plan':
        company_id = request.POST.get('company_id', '').strip()
        new_plan   = request.POST.get('plan', '').strip()
        if new_plan in ('starter', 'business', 'enterprise') and company_id:
            try:
                company = Company.objects.get(id=company_id)
                sub = company.subscriptions.filter(is_active=True).order_by('-valid_from').first()
                if sub:
                    sub.plan        = new_plan
                    sub.monthly_fee = Decimal(PLAN_PRICES.get(new_plan, '0'))
                    sub.save(update_fields=['plan', 'monthly_fee'])
                else:
                    CompanySubscription.objects.create(
                        company     = company,
                        plan        = new_plan,
                        valid_from  = _dt.date.today(),
                        monthly_fee = Decimal(PLAN_PRICES.get(new_plan, '0')),
                        is_active   = True,
                    )
                messages.success(request, f'Plan za "{company.name}" promijenjen u {new_plan.upper()}.')
            except Company.DoesNotExist:
                error = 'Tvrtka nije pronađena.'
        else:
            error = 'Nevažeći plan.'
        return redirect('superadmin_companies')

    # ── Create new company ────────────────────────────────────────
    if request.method == 'POST' and request.POST.get('action') == 'create':
        name = request.POST.get('name', '').strip()
        oib  = request.POST.get('oib',  '').strip()
        if not name or not oib:
            error = 'Naziv i OIB su obavezni.'
        elif Company.objects.filter(oib=oib).exists():
            error = f'Tvrtka s OIB-om {oib} već postoji.'
        else:
            company = Company.objects.create(
                name          = name,
                oib           = oib,
                address       = request.POST.get('address', ''),
                city          = request.POST.get('city', ''),
                zip_code      = request.POST.get('zip_code', ''),
                country       = request.POST.get('country', 'HR'),
                business_type = request.POST.get('business_type', 'doo'),
            )
            plan = request.POST.get('plan', '')
            if plan in ('starter', 'business', 'enterprise'):
                CompanySubscription.objects.create(
                    company     = company,
                    plan        = plan,
                    valid_from  = _dt.date.today(),
                    monthly_fee = Decimal(PLAN_PRICES.get(plan, '0')),
                    is_active   = True,
                )
            messages.success(request, f'Tvrtka "{name}" uspješno kreirana.')
            return redirect('superadmin_companies')

    # ── Build company table ───────────────────────────────────────
    rows = []
    for c in Company.objects.all().order_by('name'):
        sub        = c.subscriptions.filter(is_active=True).order_by('-valid_from').first()
        user_count = User.objects.filter(profile__company=c, is_active=True).count()
        last_login = User.objects.filter(
            profile__company=c, is_active=True
        ).aggregate(Max('last_login'))['last_login__max']
        rows.append({
            'company':    c,
            'sub':        sub,
            'user_count': user_count,
            'last_login': last_login,
            'valid_until': sub.valid_until if sub else None,
        })

    from accounts.models import ImpersonationLog
    recent_logs = ImpersonationLog.objects.select_related(
        'superadmin', 'company').order_by('-started_at')[:20]

    pending_reg_count = CompanyRegistration.objects.filter(status='pending').count()

    return render(request, 'dashboards/superadmin_companies.html', {
        'rows':              rows,
        'error':             error,
        'recent_logs':       recent_logs,
        'pending_reg_count': pending_reg_count,
    })

    branches = CompanyBranch.objects.filter(parent_company=company).order_by('name')

    if request.method == 'POST':
        action = request.POST.get('action')

        if action == 'add':
            name        = request.POST.get('name', '').strip()
            branch_type = request.POST.get('branch_type', 'other')
            address     = request.POST.get('address', '').strip()
            city        = request.POST.get('city', '').strip()
            phone       = request.POST.get('phone', '').strip()
            email       = request.POST.get('email', '').strip()
            branch_code = request.POST.get('branch_code', '').strip()
            modules_raw = request.POST.getlist('active_modules')

            if not name:
                _msg.error(request, 'Naziv podružnice je obavezan.')
            else:
                try:
                    CompanyBranch.objects.create(
                        parent_company=company,
                        name=name,
                        branch_type=branch_type,
                        address=address,
                        city=city,
                        phone=phone,
                        email=email,
                        branch_code=branch_code,
                        active_modules=modules_raw,
                    )
                    _msg.success(request, f'Podružnica "{name}" kreirana.')
                except Exception as e:
                    _msg.error(request, f'Greška: {e}')

        elif action == 'toggle':
            branch_id = request.POST.get('branch_id')
            try:
                b = CompanyBranch.objects.get(pk=branch_id, parent_company=company)
                b.is_active = not b.is_active
                b.save()
                _msg.success(request, f'{"Aktivirana" if b.is_active else "Deaktivirana"}: {b.name}')
            except CompanyBranch.DoesNotExist:
                _msg.error(request, 'Podružnica nije pronađena.')

        elif action == 'delete':
            branch_id = request.POST.get('branch_id')
            try:
                b = CompanyBranch.objects.get(pk=branch_id, parent_company=company)
                name = b.name
                b.delete()
                _msg.success(request, f'Podružnica "{name}" izbrisana.')
            except CompanyBranch.DoesNotExist:
                _msg.error(request, 'Podružnica nije pronađena.')

        elif action == 'update_modules':
            branch_id   = request.POST.get('branch_id')
            modules_raw = request.POST.getlist('active_modules')
            try:
                b = CompanyBranch.objects.get(pk=branch_id, parent_company=company)
                b.active_modules = modules_raw
                b.save()
                _msg.success(request, f'Moduli ažurirani za {b.name}.')
            except CompanyBranch.DoesNotExist:
                _msg.error(request, 'Podružnica nije pronađena.')

        return redirect('superadmin_company_branches', company_id=company_id)

    # All available modules for checkbox list
    ALL_MODULES = [
        ('invoicing',   'Fakturiranje'),
        ('pos',         'POS blagajna'),
        ('inventory',   'Zalihe'),
        ('appointments','Termini'),

        ('fitness',     'Fitness'),
        ('rental',      'Najam'),
        ('webshop',     'Webshop'),
        ('assessment',  'Procjene'),
        ('interventions','Intervencije'),
    ]

    return render(request, 'dashboards/company_branches.html', {
        'company':     company,
        'branches':    branches,
        'all_modules': ALL_MODULES,
        'branch_types': CompanyBranch.BRANCH_TYPES,

"""
Core business logic — dashboard statistics, impersonation, health checks.

Pure functions that aggregate data across modules.
No request/response dependency (except impersonation which needs session).
"""
import logging
from datetime import timedelta

from django.utils import timezone

logger = logging.getLogger(__name__)


# ── Company Dashboard Stats ─────────────────────────────────

def get_company_dashboard_stats(company, user) -> dict:
    """
    Aggregate all dashboard statistics for a company.
    Returns dict with all widget data.
    """
    today = timezone.now().date()
    week_start = today - timedelta(days=today.weekday())
    month_start = today.replace(day=1)

    stats = {
        "company": company,
        "today": today,
    }

    stats.update(_invoice_stats(company, month_start))
    stats.update(_worktime_stats(user, week_start, month_start))
    stats.update(_project_stats(company))
    stats.update(_partner_stats(company))
    stats.update(_finance_stats(company, month_start))
    stats.update(_appointment_stats(company, today, month_start))
    stats.update(_pos_stats(today))

    return stats


def _invoice_stats(company, month_start) -> dict:
    """Invoice aggregation for dashboard."""
    try:
        from invoicing.models import Invoice
        qs = Invoice.objects.all()
        if company:
            qs = qs.filter(company=company)

        return {
            "total_invoiced": qs.filter(
                doctype="R", status__in=("SENT", "PAID", "UNPAID", "OVERDUE")
            ).aggregate(s=__import__("django.db.models", fromlist=["Sum"]).Sum("total"))["s"] or 0,
            "unpaid": qs.filter(
                doctype="R", status__in=("UNPAID", "OVERDUE")
            ).aggregate(s=__import__("django.db.models", fromlist=["Sum"]).Sum("total"))["s"] or 0,
            "paid_month": qs.filter(
                doctype="R", status="PAID", date_issued__date__gte=month_start
            ).aggregate(s=__import__("django.db.models", fromlist=["Sum"]).Sum("total"))["s"] or 0,
            "recent_invoices": list(qs.filter(doctype="R").select_related("company").order_by("-date_issued")[:8]),
            "pending_offers": qs.filter(doctype="PO", status__in=("DRAFT", "SENT")).count(),
        }
    except Exception as e:
        logger.warning("Invoice stats failed: %s", e)
        return {"total_invoiced": 0, "unpaid": 0, "paid_month": 0, "recent_invoices": [], "pending_offers": 0}


def _worktime_stats(user, week_start, month_start) -> dict:
    """Worktime aggregation."""
    try:
        from django.db.models import Sum

        from worktime.models import TimeEntry

        qs = TimeEntry.objects.filter(user=user)
        return {
            "hours_week": qs.filter(date__gte=week_start).aggregate(s=Sum("duration"))["s"] or 0,
            "hours_month": qs.filter(date__gte=month_start).aggregate(s=Sum("duration"))["s"] or 0,
            "recent_entries": list(qs.select_related("project", "user").order_by("-date")[:5]),
        }
    except Exception as e:
        logger.warning("Worktime stats failed: %s", e)
        return {"hours_week": 0, "hours_month": 0, "recent_entries": []}


def _project_stats(company) -> dict:
    """Project aggregation."""
    try:
        from projects.models import Project
        qs = Project.objects.all()
        if company:
            qs = qs.filter(company=company)
        return {
            "projects": list(qs.filter(status__in=("active", "in_progress", "open")).select_related("company").order_by("-id")[:5]),
            "total_projects": qs.exclude(status__in=("completed", "cancelled")).count(),
        }
    except Exception as e:
        logger.warning("Project stats failed: %s", e)
        return {"projects": [], "total_projects": 0}


def _partner_stats(company) -> dict:
    """Partner count."""
    try:
        from partners.models import Partner
        qs = Partner.objects.all()
        if company:
            qs = qs.filter(company=company)
        return {"total_partners": qs.count()}
    except Exception as e:
        logger.warning("Partner stats failed: %s", e)
        return {"total_partners": 0}


def _finance_stats(company, month_start) -> dict:
    """Finance aggregation."""
    try:
        from django.db.models import Sum

        from finance.models import Transaction

        qs = Transaction.objects.all()
        if company:
            qs = qs.filter(company=company)

        income = qs.filter(direction="income", date__gte=month_start).aggregate(s=Sum("amount"))["s"] or 0
        expense = qs.filter(direction="expense", date__gte=month_start).aggregate(s=Sum("amount"))["s"] or 0
        return {
            "income_month": income,
            "expense_month": expense,
            "cashflow_month": income - expense,
        }
    except Exception as e:
        logger.warning("Finance stats failed: %s", e)
        return {"income_month": 0, "expense_month": 0, "cashflow_month": 0}


def _appointment_stats(company, today, month_start) -> dict:
    """Appointment aggregation."""
    try:
        from appointments.models import Appointment

        qs = Appointment.objects.filter(start_time__date=today)
        if company:
            qs = qs.filter(company=company)
        month_count = Appointment.objects.filter(start_time__date__gte=month_start)
        if company:
            month_count = month_count.filter(company=company)

        return {
            "today_appointments": list(qs.select_related("client", "service", "resource").order_by("start_time")[:8]),
            "total_appointments_month": month_count.count(),
        }
    except Exception as e:
        logger.warning("Appointment stats failed: %s", e)
        return {"today_appointments": [], "total_appointments_month": 0}


def _pos_stats(today) -> dict:
    """POS revenue today."""
    try:
        from django.db.models import Sum

        from pos.models import POSReceipt

        qs = POSReceipt.objects.filter(created_at__date=today, status="paid")
        return {
            "pos_revenue_today": qs.aggregate(s=Sum("total"))["s"] or 0,
            "pos_count_today": qs.count(),
        }
    except Exception as e:
        logger.warning("POS stats failed: %s", e)
        return {"pos_revenue_today": 0, "pos_count_today": 0}


# ── Superadmin Stats ────────────────────────────────────────

def get_superadmin_stats() -> dict:
    """Aggregate platform-wide statistics for superadmin."""
    from django.contrib.auth.models import User

    from companies.models import Company

    try:
        return {
            "total_users": User.objects.filter(is_active=True).count(),
            "total_companies": Company.objects.count(),
            "active_companies": Company.objects.filter(is_active=True).count(),
            "companies_list": list(
                Company.objects.all()
                .order_by("name")
                .values("id", "name", "oib", "is_active", "created_at")[:50]
            ),
        }
    except Exception as e:
        logger.exception("Superadmin stats failed: %s", e)
        return {"total_users": 0, "total_companies": 0, "active_companies": 0, "companies_list": []}


# ── Impersonation ───────────────────────────────────────────

def start_impersonation(admin_user, target_company_id: int) -> dict:
    """
    Start impersonating a company. Returns session data to set.
    Raises PermissionError if user is not superadmin.
    """
    if not admin_user.is_staff:
        raise PermissionError("Samo administratori mogu koristiti impersonaciju.")

    from companies.models import Company
    target = Company.objects.get(pk=target_company_id)

    logger.info("Impersonation started: admin=%s → company=%s", admin_user.pk, target.name)

    return {
        "impersonate_company_id": target.pk,
        "impersonate_company_name": target.name,
        "original_company_id": getattr(admin_user, "profile", None)
        and admin_user.profile.company_id,
    }


def stop_impersonation(admin_user) -> None:
    """Stop impersonating. Logs the action."""
    logger.info("Impersonation stopped by admin=%s", admin_user.pk)
